-- Grain: one row = one ticket on one flight (one booking)
-- i.e. every row in the fact table is a single passenger's single booking for a single flight

Dimension keys (from bronze_bookings):
- booking_id -> degenerate dimension, identifies the row itself, no separate dim table for it
- passenger_id -> who, links to dim_passenger
- flight_id -> what, links to dim_flight
- booking_date -> when, links to dim_date (role: date booked)
- travel_date -> when, links to dim_date (role: date traveled)
- fare_class -> what, Economy/Business etc, links to dim_fare_class
- booking_status -> what, Confirmed/Cancelled etc, links to dim_booking_status

Dimension keys (from bronze_flights, joined in):
- flight_number -> what, which dim_flight
- origin_airport_code -> where, links to dim_airport (role: origin)
- dest_airport_code -> where, links to dim_airport (role: destination)
- aircraft_code -> what, links to dim_aircraft
- flight_date -> when, links to dim_date (role: flight date)

Measures (from bronze_bookings):
- fare_amount
- tax_amount
- miles_earned

Additive vs non-additive: 

Additive: Fare_amount, tax_amount and miles_earned are all additive, they can be summed across passengers, routes, dates, fare class etc and the total still means something (e.g total revenue). 

Non-Additive: A non-additive example would be something like fare per mile (fare_amount / miles_earned), summing that ratio across bookings gives a meaningless number, you'd have to sum fare_amount and miles_earned separately first and then divide.

-- Change EffectiveFrom from nullable to NOT NULL
-- This is done after populating all existing NULL values
IF EXISTS (SELECT 1 FROM sys.columns WHERE object_id = OBJECT_ID('dw.DimPassenger') AND name = 'EffectiveFrom' AND is_nullable = 1)

-- Variable to store the name of the existing unique constraint
DECLARE @uqName sysname;

-- Find the unique constraint defined on PassengerId
SELECT @uqName = kc.name
FROM sys.key_constraints kc
JOIN sys.tables  t   ON kc.parent_object_id = t.object_id
JOIN sys.schemas s   ON t.schema_id = s.schema_id
JOIN sys.index_columns ic ON ic.object_id = t.object_id AND ic.index_id = kc.unique_index_id
JOIN sys.columns c   ON c.object_id = t.object_id AND c.column_id = ic.column_id
WHERE s.name = 'dw' AND t.name = 'DimPassenger' AND kc.type = 'UQ' AND c.name = 'PassengerId';

-- Drop the existing unique constraint if it exists
IF @uqName IS NOT NULL
    EXEC('ALTER TABLE dw.DimPassenger DROP CONSTRAINT [' + @uqName + ']');
GO

-- Create a unique filtered index on PassengerId
-- This allows only one current version (IsCurrent = 1)
-- Multiple historical versions can exist for the same PassengerId
IF NOT EXISTS (SELECT 1 FROM sys.indexes WHERE object_id = OBJECT_ID('dw.DimPassenger') AND name = 'UQ_DimPassenger_CurrentVersion')
    CREATE UNIQUE NONCLUSTERED INDEX UQ_DimPassenger_CurrentVersion
    ON dw.DimPassenger (PassengerId)
    WHERE IsCurrent = 1;
GO

-- =======================================================
-- SCD2 Load
-- =======================================================

DECLARE @Today DATE = CAST(GETDATE() AS DATE);
DECLARE @MergeOut TABLE (Action NVARCHAR(10), PassengerId BIGINT);

BEGIN TRANSACTION;

    -- PASS 1: expire changed current rows, insert brand-new passengers
    MERGE dw.DimPassenger AS tgt
    USING dbo.stg_passenger_updates AS src
        ON tgt.PassengerId = src.passenger_id
       AND tgt.IsCurrent   = 1
    WHEN MATCHED AND (
            ISNULL(tgt.FrequentFlyerTier, N'') <> ISNULL(src.frequent_flyer_tier, N'')
         OR ISNULL(tgt.HomeAirportCode,   N'') <> ISNULL(src.home_airport_code,   N'')
         )
        THEN UPDATE SET
            tgt.IsCurrent   = 0,
            tgt.EffectiveTo = DATEADD(DAY, -1, @Today)
    WHEN NOT MATCHED BY TARGET THEN
        INSERT (PassengerId, PassengerName, HomeAirportCode, FrequentFlyerTier,
                SignupDate, EffectiveFrom, EffectiveTo, IsCurrent)
        VALUES (src.passenger_id, src.passenger_name, src.home_airport_code, src.frequent_flyer_tier,
                NULL, @Today, NULL, 1)
    OUTPUT $action, COALESCE(inserted.PassengerId, deleted.PassengerId)
    INTO @MergeOut (Action, PassengerId);

    -- PASS 2: open a new current version for every passenger expired above
    INSERT INTO dw.DimPassenger
        (PassengerId, PassengerName, HomeAirportCode, FrequentFlyerTier,
         SignupDate, EffectiveFrom, EffectiveTo, IsCurrent)
    SELECT
        s.passenger_id, s.passenger_name, s.home_airport_code, s.frequent_flyer_tier,
        prior.SignupDate, @Today, NULL, 1
    FROM dbo.stg_passenger_updates s
    INNER JOIN @MergeOut m
        ON m.PassengerId = s.passenger_id
       AND m.Action = 'UPDATE'
    INNER JOIN dw.DimPassenger prior
        ON prior.PassengerId = s.passenger_id
       AND prior.IsCurrent   = 0
       AND prior.EffectiveTo = DATEADD(DAY, -1, @Today);

COMMIT TRANSACTION;
GO
