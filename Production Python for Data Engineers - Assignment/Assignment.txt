    def total(self):
        # sum this order's items, then apply the discount
        total = 0
        for name, price in self.items:
            total = total + clean_price(price)      # reuse task 1's cleaner

"""
legacy_orders.py

import csv
import logging
import os
import time

from dotenv import load_dotenv      # reads the .env file into the environment

from gateway import save_to_orders_api   # the flaky API client we're calling

LOG_FILE = "orders.log"
MAX_ATTEMPTS = 3            # how many times to try saving one order
RETRY_DELAY_SECONDS = 1     # short pause between attempts

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(message)s",
    datefmt="%Y-%m-%d %H:%M:%S",
    handlers=[
        logging.FileHandler(LOG_FILE, encoding="utf-8"),
        logging.StreamHandler(),
    ],
)
logger = logging.getLogger("legacy_orders")

load_dotenv()  # pulls LOAFLY_API_KEY (and anything else) in from .env, if present
API_KEY = os.getenv("LOAFLY_API_KEY")  # never hardcode this - read it from env


def load_rows(path):
    # this one really should stop the run if it fails - no point
    # continuing with no data at all
    try:
        with open(path, newline="", encoding="utf-8") as f:
            rows = list(csv.DictReader(f))
    except OSError:
        logger.error("couldn't open %s", path, exc_info=True)
        raise
    logger.info("loaded %d rows from %s", len(rows), path)
    return rows


def group_by_order(rows):
    # multiple csv rows -> one dict per order_id
    orders = {}
    for row in rows:
        oid = row["order_id"]
        if oid not in orders:
            orders[oid] = {"customer": row["customer"], "items": []}
        orders[oid]["items"].append((row["item_name"], row["item_price"]))
    logger.info("grouped into %d orders", len(orders))
    return orders


def parse_price(raw_price):
    return float(raw_price.replace(",", ""))


def save_with_retry(order_id, total):
    for attempt in range(1, MAX_ATTEMPTS + 1):
        try:
            return save_to_orders_api(order_id, total)
        except ConnectionError as e:
            logger.warning(
                "order %s: save attempt %d/%d failed - %s",
                order_id, attempt, MAX_ATTEMPTS, e,
            )
            if attempt < MAX_ATTEMPTS:
                time.sleep(RETRY_DELAY_SECONDS)


def process_orders(orders):
    processed = 0
    skipped_items = 0
    failed_saves = 0

    for oid, order in orders.items():
        total = 0.0
        for name, price in order["items"]:
            try:
                item_total = parse_price(price)
            except (ValueError, AttributeError):
                logger.warning("order %s: skipping '%s', bad price %r", oid, name, price)
                skipped_items += 1
                continue
            finally:
                pass  # nothing to clean up, just here per the assignment spec
            total += item_total

        total -= total * 10 / 100  # 10% discount

        result = save_with_retry(oid, total)
        if result is None:
            failed_saves += 1
        else:
            logger.info("saved order %s for %s - total %.2f", oid, order["customer"], total)
        processed += 1

    logger.info(
        "done: %d orders processed, %d items skipped, %d saves failed",
        processed, skipped_items, failed_saves,
    )


def main():
    if not API_KEY:
        logger.error("LOAFLY_API_KEY is not set - copy env.example to .env and fill it in")
        raise SystemExit(1)
    rows = load_rows("raw_orders.csv")
    orders = group_by_order(rows)
    process_orders(orders)


if __name__ == "__main__":
    main()
